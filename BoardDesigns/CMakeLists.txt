cmake_minimum_required(VERSION 3.15)

# Normally configured on its own -- cmake -S BoardDesigns -B build-boards -- so
# it needs a project() of its own. NONE because nothing here is compiled; it
# only runs kicad-cli and a couple of scripts, and asking CMake to find a C
# compiler would slow the configure down for nothing.
project(BoardDocs NONE)

# This file produces the figures the documentation uses: a 3D render of each
# board, and its schematic as a PDF.
#
# Both are made with kicad-cli, which ships inside KiCad, so nothing needs
# installing beyond KiCad itself -- plus Pillow, for the trimming step. They are
# written into the documentation tree and committed, because the workflow that
# publishes the hardware site installs Python and MkDocs only; it has no KiCad
# to generate with. That is also why each board carries a stamp of what it was
# generated from, so CI can still tell when a figure has gone stale.
#
# Fabrication outputs are no longer driven from here. They were generated by
# KiBot, an external tool, through an add_pcb_outputs() function that every
# board directory called; that path is retired. The .kibot.yaml files remain in
# the tree to run by hand if wanted.

find_program(kicad_cli_EXECUTABLE kicad-cli
             HINTS /Applications/KiCad/KiCad.app/Contents/MacOS/
             NO_CACHE)

# A hard error, because the alternative is worse than a failure. Without
# kicad-cli the functions below used to return early, so configuring succeeded,
# no targets were created, and a build asked to produce every figure quietly
# produced none and reported success. A CI run did exactly that.
if(NOT kicad_cli_EXECUTABLE)
  message(FATAL_ERROR
    "kicad-cli not found. It ships inside KiCad; on macOS it lives in "
    "KiCad.app/Contents/MacOS, which is already searched. Point at it with "
    "-Dkicad_cli_EXECUTABLE=/path/to/kicad-cli if it is somewhere else.")
endif()
message(STATUS "kicad-cli: ${kicad_cli_EXECUTABLE}")

# Renders are committed, so they belong in the source tree, not the build tree.
get_filename_component(_hw_root "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(BOARD_RENDER_DIR "${_hw_root}/docs/src/images/boards")

# Beside each set of renders, a stamp holding the SHA-256 of the .kicad_pcb they
# were drawn from. CMake uses file timestamps, which a clone does not preserve,
# so a checkout cannot tell by itself whether its images are current. The stamps
# let CI answer that question without KiCad: hash the board, compare. They live
# outside the documentation tree so MkDocs does not publish them.
set(BOARD_STAMP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.render-stamps")
set(BOARD_STAMP_SCRIPT "${CMAKE_CURRENT_SOURCE_DIR}/kicad-helpers/stamp.cmake")
set(BOARD_FIT_SCRIPT   "${CMAKE_CURRENT_SOURCE_DIR}/kicad-helpers/fit_render.py")

# Schematics are exported to PDF and published alongside the renders, so each
# board's page can link the drawing it describes.
set(BOARD_SCHEMATIC_DIR "${_hw_root}/docs/src/schematics")

# The renders are trimmed after drawing (see fit_render.py), which needs Pillow.
# A hard error rather than a warning: a warning scrolls past, and the build then
# spends several minutes producing exactly the badly-framed images the trim
# exists to prevent. That happened once already.
#
# find_package picks the first interpreter on the system, which on macOS is
# often /usr/bin/python3 rather than the one with your packages in it. Point it
# somewhere else with -DPython3_EXECUTABLE=/opt/homebrew/bin/python3.
find_package(Python3 COMPONENTS Interpreter REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import PIL"
                RESULT_VARIABLE _pil OUTPUT_QUIET ERROR_QUIET)
if(NOT _pil EQUAL 0)
  message(FATAL_ERROR
    "Pillow is not installed for ${Python3_EXECUTABLE}, so the board renders "
    "could not be trimmed. Install it there:\n"
    "  ${Python3_EXECUTABLE} -m pip install Pillow\n"
    "or point CMake at another interpreter with -DPython3_EXECUTABLE=...")
endif()

# Framing. Measured, after two wrong guesses, with
# kicad-helpers/probe_render_framing.py:
#
#     frame   --zoom     board px   board/frame   clipped
#      1200        1    1156x1156         0.983   -
#      1200        2    1176x1176         1.000   all four edges
#      1200      0.5      578x578         0.491   -
#      2400        1    2311x2311         0.969   -
#      2400      0.5    1156x1156         0.485   -
#
# So: the camera fits the board outline to the frame, and --zoom scales that fit
# linearly in both directions. A larger frame alone buys no margin, because the
# board grows with it -- which is why 2400 square clipped every board. --zoom
# below 1 is what pulls the camera back, and it does work with a fraction; the
# earlier conclusion that it did not came from two renders whose board widths
# coincided at 889 pixels for unrelated reasons.
#
# At zoom 1 the outline all but fills the frame, so anything overhanging it --
# a coin cell, mostly -- is cut off. 0.5 is one of the points actually measured
# above rather than an interpolation between them, and it leaves the board at
# about half the frame: a quarter of the frame clear on every side, which is
# half the board's own size again. fit_render.py trims that slack away, so it
# costs render time and nothing in the committed file.
#
# The frame is then only about resolution. At this zoom a 2400 frame puts the
# board's limiting dimension near 1180 pixels, comfortable for figures shown at
# 420.
# Plain variables, deliberately not CACHE. A cache entry is written the first
# time a build directory is configured and never updated when the default here
# changes, so an existing build-boards went on rendering at the 0.9 this file
# used to carry long after it said 0.5 -- the clipping that followed looked like
# a fault in the board. Override for an experiment with -DRENDER_ZOOM=0.4; the
# names differ from these on purpose, so nothing of ours can go stale in a
# cache.
set(BOARD_RENDER_ZOOM "0.5")
set(BOARD_RENDER_FRAME "2400")
set(_render_note "")
if(DEFINED RENDER_ZOOM)
  set(BOARD_RENDER_ZOOM "${RENDER_ZOOM}")
  set(_render_note " (zoom overridden)")
endif()
if(DEFINED RENDER_FRAME)
  set(BOARD_RENDER_FRAME "${RENDER_FRAME}")
  set(_render_note "${_render_note} (frame overridden)")
endif()

# Printed because the value in force was invisible, which is what let a stale
# one go unnoticed through two builds. An override is named as such, because a
# -D sticks in the cache for every later reconfigure of that directory and is
# otherwise just as quiet as the thing it replaced.
message(STATUS
  "Board renders: zoom ${BOARD_RENDER_ZOOM}, frame ${BOARD_RENDER_FRAME} "
  "square${_render_note}")

# Aggregate target: "cmake --build . --target board-renders" redraws every
# board whose .kicad_pcb has changed since its images were written. Deliberately
# not part of ALL -- a normal build should not rewrite files under version
# control.
add_custom_target(board-renders
  COMMENT "Board renders for the documentation")

# Schematic PDFs, on the same footing: "cmake --build . --target board-schematics".
# "board-docs" does both, which is what you usually want.
add_custom_target(board-schematics
  COMMENT "Schematic PDFs for the documentation")
add_custom_target(board-docs)
add_dependencies(board-docs board-renders board-schematics)

# Renders for one board, regenerated when its layout changes. The sides to draw
# follow the name: tags are documented from both faces because components sit on
# each, while the bases and prototype carriers are single-sided in any way that
# matters and only need a top view.
function(_board_renders name)
  set(board "${CMAKE_CURRENT_SOURCE_DIR}/${name}.kicad_pcb")
  set(outputs "")
  set(images "")

  foreach(side ${ARGN})
    set(out "${BOARD_RENDER_DIR}/${name}-${side}.png")
    set(raw "${CMAKE_CURRENT_BINARY_DIR}/${name}-${side}.raw.png")
    set(done "${CMAKE_CURRENT_BINARY_DIR}/${name}-${side}.done")

    # The declared output is a marker in the build tree, not the image itself.
    # make deletes an output whose command failed, and the images are committed
    # files: a single board that would not trim once wiped all nineteen of them.
    # With the marker as the output there is nothing make can destroy, and the
    # two-step render-then-trim means a failure leaves the previous image in
    # place rather than a half-made one.
    add_custom_command(
      OUTPUT "${done}"
      COMMAND ${CMAKE_COMMAND} -E make_directory "${BOARD_RENDER_DIR}"
      # --background transparent so one image reads correctly against both the
      # light and dark documentation themes, and so the trim can find the board.
      # Long option names throughout: kicad-cli's short -h means height, not
      # help.
      COMMAND ${kicad_cli_EXECUTABLE} pcb render
              --side ${side}
              --rotate 0,0,90
              --zoom ${BOARD_RENDER_ZOOM}
              --quality high
              --width ${BOARD_RENDER_FRAME}
              --height ${BOARD_RENDER_FRAME}
              --background transparent
              --output "${raw}"
              "${board}"
      COMMAND "${Python3_EXECUTABLE}" "${BOARD_FIT_SCRIPT}" "${raw}"
              --output "${out}"
      COMMAND ${CMAKE_COMMAND} -E touch "${done}"
      DEPENDS "${board}" "${BOARD_FIT_SCRIPT}"
      COMMENT "Rendering ${name} (${side})"
      VERBATIM)

    list(APPEND outputs "${done}")
    list(APPEND images "${out}")
  endforeach()

  # One stamp per board, written after its images, so a half-finished render
  # never leaves a stamp claiming the board is up to date. Separate from the
  # render commands because a two-sided board has two of them and they would
  # otherwise race to write the same file.
  set(stamp "${BOARD_STAMP_DIR}/${name}.sha256")
  add_custom_command(
    OUTPUT "${stamp}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${BOARD_STAMP_DIR}"
    COMMAND ${CMAKE_COMMAND} -DBOARD=${board} -DSTAMP=${stamp} -P "${BOARD_STAMP_SCRIPT}"
    DEPENDS ${outputs}
    COMMENT "Stamping ${name}"
    VERBATIM)

  add_custom_target(${name}-renders DEPENDS ${outputs} "${stamp}")
  add_dependencies(board-renders ${name}-renders)
endfunction()

# The schematic as a PDF, exported straight from the drawing the board was laid
# out from, so a page can link the real thing rather than a transcription of it.
# A hierarchical design exports every sheet into the one file.
function(_board_schematic name)
  set(sch "${CMAKE_CURRENT_SOURCE_DIR}/${name}.kicad_sch")
  if(NOT EXISTS "${sch}")
    message(WARNING "No ${name}.kicad_sch -- skipping its schematic PDF")
    return()
  endif()

  # A sub-sheet edit changes the drawing without touching the root file, so the
  # whole set is the dependency. Globbed at configure time: adding a sheet means
  # re-running cmake, which adding a sheet already does.
  file(GLOB sheets CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.kicad_sch")

  set(pdf "${BOARD_SCHEMATIC_DIR}/${name}.pdf")
  add_custom_command(
    OUTPUT "${pdf}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${BOARD_SCHEMATIC_DIR}"
    # --no-background-color so the PDF prints on white whatever colour scheme
    # the drawing is edited in.
    COMMAND ${kicad_cli_EXECUTABLE} sch export pdf
            --no-background-color
            --output "${pdf}"
            "${sch}"
    DEPENDS ${sheets}
    COMMENT "Exporting ${name} schematic"
    VERBATIM)

  # Stamped against every sheet, so the check can tell a published PDF that is
  # of an older drawing.
  set(stamp "${BOARD_STAMP_DIR}/${name}.sch.sha256")
  add_custom_command(
    OUTPUT "${stamp}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${BOARD_STAMP_DIR}"
    COMMAND ${CMAKE_COMMAND} "-DBOARD=${sheets}" -DSTAMP=${stamp} -P "${BOARD_STAMP_SCRIPT}"
    DEPENDS "${pdf}"
    COMMENT "Stamping ${name} schematic"
    VERBATIM)

  add_custom_target(${name}-schematic DEPENDS "${pdf}" "${stamp}")
  add_dependencies(board-schematics ${name}-schematic)
endfunction()

# Both faces -- for the tag boards, which carry parts on each side.
function(add_board_renders name)
  _board_renders(${name} top bottom)
  _board_schematic(${name})
endfunction()

# Top only -- for bases and prototype carriers.
function(add_board_renders_top name)
  _board_renders(${name} top)
  _board_schematic(${name})
endfunction()
